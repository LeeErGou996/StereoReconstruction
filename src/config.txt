# StereoReconstruction 配置文件示例
# 支持 key=value 格式，注释以#开头

# ========== 全局重建模式 ==========
# 重建模式: 0=点云, 1=三角网格, 2=Poisson表面重建
# 选择不同的重建方式，影响最终输出类型
reconstructionMode=2

# ========== 视差计算参数 ==========
# numDisparities: 最大视差范围（必须为16的倍数），影响深度精度和计算速度。推荐128~256。
numDisparities=256
# blockSize: 匹配窗口大小（奇数），越大越平滑但细节损失，越小细节多但噪声大。推荐5~15。
blockSize=11

# ========== 特征点算法 ==========
# algorithm: 特征点提取算法，支持 ORB/SIFT/SURF。ORB速度快，SIFT/SURF更鲁棒但慢。
algorithm=SIFT

# ========== 路径参数 ==========
# leftDir/rightDir: 左右图像文件夹路径（末尾加/）
# cameraDir: 相机内参文件夹路径
# outputDir: 输出结果文件夹路径
leftDir=../data/left/
rightDir=../data/right/
cameraDir=../data/camera/
outputDir=../test/

# ========== 相机畸变参数 ==========
# distCoeffs: 相机畸变系数（k1,k2,p1,p2,k3），通常为5个浮点数。无畸变可全0。
distCoeffs=0,0,0,0,0

# ========== Mesh/Poisson重建参数 ==========
# meshFormat: 网格保存格式（ply/obj等）
meshFormat=ply
# triangulationStep: 三角化步长，越大点云稀疏，越小更密集但慢。推荐2~10。
triangulationStep=5
# maxDepthDifference: 三角化时允许的最大深度差，防止跨越大深度跳变。推荐50~200。
maxDepthDifference=100.0
# useColor: 网格是否带颜色（1=带色，0=无色）
useColor=1
# depthThreshold: 参与重建的最大深度阈值，过滤远处噪声点。单位与深度一致，推荐1000~10000。
depthThreshold=5000.0

# ========== denseMatching/SGBM参数 ==========
# sgbmP1: SGBM平滑项P1，较小值更敏感，推荐1-10。P1变大，视差图更平滑但细节丢失，变小则噪声增多但细节更丰富。
sgbmP1=1
# sgbmP2: SGBM平滑项P2，较大值更平滑，推荐4-40。P2变大，能抑制错误匹配但可能丢失边缘，变小则边缘更锐利但易出错。
sgbmP2=4
# preFilterCap: 预滤波器上限，推荐15-63。变大可增强对比度但易引入伪影，变小则对比度低但更平滑。
preFilterCap=31
# uniquenessRatio: 唯一性比率，越大越严格，推荐5-50。变大可减少误匹配但可能导致更多无效区域，变小则视差更密集但误差增多。
uniquenessRatio=50
# speckleWindowSize: 孤立斑点窗口大小，推荐0-200。变大可去除更多小噪点但可能误删细节，变小则噪点增多。
speckleWindowSize=100
# speckleRange: 孤立斑点视差范围，推荐0-32。变大可去除更多异常视差但可能误删，变小则保留更多细节但噪点增多。
speckleRange=8
# disp12MaxDiff: 左右一致性最大容差，推荐1-10。变大可保留更多视差但误差增多，变小则更严格但无效区域增多。
disp12MaxDiff=5


# ========== Poisson重建高级参数 ==========
# poissonDepth: 八叉树深度，越大mesh越精细，推荐8~12，默认8
poissonDepth=12
# poissonSolverDivide: 求解器分割深度，通常与poissonDepth一致
poissonSolverDivide=8
# poissonSamplesPerNode: 每节点采样点数，影响细节和噪声，推荐1.0~3.0
poissonSamplesPerNode=3
# poissonFullDepth: 完全重建深度，影响大尺度结构，推荐4~8
poissonFullDepth=8.0
# poissonTrim: mesh裁剪参数，0为不裁剪，>0可去除边缘伪影
poissonTrim=0.0
# poissonUseConfidence: 是否使用置信度权重，true/false
poissonUseConfidence=false
# poissonManifold: 是否保持流形，true/false
poissonManifold=false
# poissonOutputPolygons: 是否输出多边形，true/false
poissonOutputPolygons=false

# depthDiffThreshold: 法线估计和三角化的深度差异阈值系数，推荐0.05~0.2
# 控制法线估计时允许的深度跳变比例，越大越宽松
# 例如0.1表示允许10%的深度跳变
# normalNeighbors: 法线估计时的邻居点数，推荐10~30
# 越大法线更平滑，越小更敏感

depthDiffThreshold=0.05
normalNeighbors=15
