// ==================== 配置参数 ====================
struct PoissonConfig {
    // ========== Poisson重建高级参数 ==========
    // poissonDepth: 八叉树深度，越大mesh越精细，推荐8~12，默认8
    int poissonDepth = 13;
    // poissonSolverDivide: 求解器分割深度，通常与poissonDepth一致
    int poissonSolverDivide = 14;
    // poissonSamplesPerNode: 每节点采样点数，影响细节和噪声，推荐1.0~3.0
    float poissonSamplesPerNode = 3.0f;
    // poissonFullDepth: 完全重建深度，影响大尺度结构，推荐4~8
    int poissonFullDepth = 10;
    // poissonTrim: mesh裁剪参数，0为不裁剪，>0可去除边缘伪影
    float poissonTrim = 0.0f;
    // poissonUseConfidence: 是否使用置信度权重
    bool poissonUseConfidence = false;
    // poissonManifold: 是否保持流形
    bool poissonManifold = false;
    // poissonOutputPolygons: 是否输出多边形
    bool poissonOutputPolygons = false;
    
    // ========== 深度和法线参数 ==========
    // depthDiffThreshold: 法线估计和三角化的深度差异阈值系数，推荐0.05~0.2
    // 控制法线估计时允许的深度跳变比例，越大越宽松
    // 例如0.1表示允许10%的深度跳变
    float depthDiffThreshold = 0.1f;
    // normalNeighbors: 法线估计时的邻居点数，推荐10~30
    // 越大法线更平滑，越小更敏感
    int normalNeighbors = 10;
    
    // ========== Mesh/Poisson重建参数 ==========
    // meshFormat: 网格保存格式（ply/obj等）
    std::string meshFormat = "ply";
    // triangulationStep: 三角化步长，越大点云稀疏，越小更密集但慢。推荐2~10。
    int triangulationStep = 2;
    // maxDepthDifference: 三角化时允许的最大深度差，防止跨越大深度跳变。推荐50~200。
    float maxDepthDifference = 20.0f;
    // useColor: 网格是否带颜色（1=带色，0=无色）
    bool useColor = true;
    // depthThreshold: 参与重建的最大深度阈值，过滤远处噪声点。单位与深度一致，推荐1000~10000。
    float depthThreshold = 5000.0f;
    
    // ========== 基础参数 ==========
    float voxelSize = 0.01f;
    int sdfNeighbors = 8;
    bool useParallel = true;
    bool enableDebug = false;
    
    // SDF参数
    float sdfSearchRadius = 0.05f;
    float sdfSigma = 0.02f;
    
    // 网格参数
    int maxGridSize = 256;
    float isoValue = 0.0f;
    
    static PoissonConfig& instance() {
        static PoissonConfig config;
        return config;
    }
    
    // 打印当前配置
    void printConfig() {
        std::cout << "\n==================== Poisson重建参数配置 ====================" << std::endl;
        std::cout << "[Poisson高级参数]" << std::endl;
        std::cout << "  八叉树深度(poissonDepth): " << poissonDepth << std::endl;
        std::cout << "  求解器分割深度(poissonSolverDivide): " << poissonSolverDivide << std::endl;
        std::cout << "  每节点采样数(poissonSamplesPerNode): " << poissonSamplesPerNode << std::endl;
        std::cout << "  完全重建深度(poissonFullDepth): " << poissonFullDepth << std::endl;
        std::cout << "  裁剪参数(poissonTrim): " << poissonTrim << std::endl;
        std::cout << "  使用置信度(poissonUseConfidence): " << (poissonUseConfidence ? "是" : "否") << std::endl;
        std::cout << "  保持流形(poissonManifold): " << (poissonManifold ? "是" : "否") << std::endl;
        std::cout << "  输出多边形(poissonOutputPolygons): " << (poissonOutputPolygons ? "是" : "否") << std::endl;
        
        std::cout << "\n[深度和法线参数]" << std::endl;
        std::cout << "  深度差异阈值(depthDiffThreshold): " << depthDiffThreshold << std::endl;
        std::cout << "  法线邻居数(normalNeighbors): " << normalNeighbors << std::endl;
        
        std::cout << "\n[网格重建参数]" << std::endl;
        std::cout << "  网格格式(meshFormat): " << meshFormat << std::endl;
        std::cout << "  三角化步长(triangulationStep): " << triangulationStep << std::endl;

// 简单的3D空间哈希表，用于快速邻域搜索
class SpatialHash {
private:
    float cellSize;
    std::unordered_map<int64_t, std::vector<int>> hashTable;
    
    int64_t hashKey(float x, float y, float z) const {
        int ix = static_cast<int>(std::floor(x / cellSize));
        int iy = static_cast<int>(std::floor(y / cellSize));
        int iz = static_cast<int>(std::floor(z / cellSize));
        return (static_cast<int64_t>(ix) << 42) | 
               (static_cast<int64_t>(iy) << 21) | 
               static_cast<int64_t>(iz);
    }
    
public:
    SpatialHash(float cell_size) : cellSize(cell_size) {}
    
    void insert(int index, float x, float y, float z) {
        int64_t key = hashKey(x, y, z);
        hashTable[key].push_back(index);
    }
    
    std::vector<int> queryRadius(float x, float y, float z, float radius) const {
        std::vector<int> result;
        int steps = static_cast<int>(std::ceil(radius / cellSize));
        
        for (int dx = -steps; dx <= steps; ++dx) {
            for (int dy = -steps; dy <= steps; ++dy) {
                for (int dz = -steps; dz <= steps; ++dz) {
                    int64_t key = hashKey(x + dx * cellSize, y + dy * cellSize, z + dz * cellSize);
                    auto it = hashTable.find(key);
                    if (it != hashTable.end()) {
                        result.insert(result.end(), it->second.begin(), it->second.end());
                    }
                }
            }
        }
        return result;
    }
};

// 简单的3x3矩阵类
struct Matrix3x3 {
    float data[3][3];
    
    Matrix3x3() {
        for (int i = 0; i < 3; ++i)
            for (int j = 0; j < 3; ++j)
                data[i][j] = 0.0f;
    }
    
    void setZero() {
        for (int i = 0; i < 3; ++i)
            for (int j = 0; j < 3; ++j)
                data[i][j] = 0.0f;
    }
    
    Matrix3x3& operator+=(const Matrix3x3& other) {
        for (int i = 0; i < 3; ++i)
            for (int j = 0; j < 3; ++j)
                data[i][j] += other.data[i][j];
        return *this;
    }
    
    Matrix3x3& operator/=(float scalar) {
        for (int i = 0; i < 3; ++i)
            for (int j = 0; j < 3; ++j)
                data[i][j] /= scalar;
        return *this;
    }
};

// Jacobi特征值分解（用于3x3对称矩阵）
void jacobiEigenDecomposition(const Matrix3x3& A, float eigenValues[3], float eigenVectors[3][3]) {
    // 复制矩阵
    Matrix3x3 matrix = A;
    
    // 初始化特征向量为单位矩阵
    for (int i = 0; i < 3; ++i) {
        for (int j = 0; j < 3; ++j) {
            eigenVectors[i][j] = (i == j) ? 1.0f : 0.0f;
        }
    }
    
    const int maxIterations = 50;
    const float tolerance = 1e-6f;
    
    for (int iter = 0; iter < maxIterations; ++iter) {
        // 找到最大的非对角元素
        float maxOffDiag = 0.0f;
        int p = 0, q = 1;
        
        for (int i = 0; i < 3; ++i) {
            for (int j = i + 1; j < 3; ++j) {
                if (std::abs(matrix.data[i][j]) > maxOffDiag) {
                    maxOffDiag = std::abs(matrix.data[i][j]);
                    p = i;
                    q = j;
                }
            }
        }
        
        if (maxOffDiag < tolerance) break;
        
        // 计算Jacobi旋转参数
        float theta;
        if (std::abs(matrix.data[p][p] - matrix.data[q][q]) < tolerance) {
            theta = M_PI / 4.0f;
        } else {
            theta = 0.5f * std::atan(2.0f * matrix.data[p][q] / (matrix.data[p][p] - matrix.data[q][q]));
        }
        
        float cosTheta = std::cos(theta);
        float sinTheta = std::sin(theta);
        
        // 应用旋转
        Matrix3x3 newMatrix = matrix;
        newMatrix.data[p][p] = cosTheta * cosTheta * matrix.data[p][p] + 
                               sinTheta * sinTheta * matrix.data[q][q] - 
                               2.0f * cosTheta * sinTheta * matrix.data[p][q];
        newMatrix.data[q][q] = sinTheta * sinTheta * matrix.data[p][p] + 
                               cosTheta * cosTheta * matrix.data[q][q] + 
                               2.0f * cosTheta * sinTheta * matrix.data[p][q];
        newMatrix.data[p][q] = newMatrix.data[q][p] = 0.0f;
        
        // 更新其他元素
        for (int i = 0; i < 3; ++i) {
            if (i != p && i != q) {
                newMatrix.data[i][p] = newMatrix.data[p][i] = 
                    cosTheta * matrix.data[i][p] - sinTheta * matrix.data[i][q];
                newMatrix.data[i][q] = newMatrix.data[q][i] = 
                    sinTheta * matrix.data[i][p] + cosTheta * matrix.data[i][q];
            }
        }
        
        matrix = newMatrix;
        
        // 更新特征向量
        for (int i = 0; i < 3; ++i) {
            float temp_p = eigenVectors[i][p];
            float temp_q = eigenVectors[i][q];
            eigenVectors[i][p] = cosTheta * temp_p - sinTheta * temp_q;
            eigenVectors[i][q] = sinTheta * temp_p + cosTheta * temp_q;
        }
    }
    
    // 提取特征值
    eigenValues[0] = matrix.data[0][0];
    eigenValues[1] = matrix.data[1][1];
    eigenValues[2] = matrix.data[2][2];
    
    // 按特征值大小排序
    for (int i = 0; i < 2; ++i) {
        for (int j = i + 1; j < 3; ++j) {
            if (eigenValues[i] > eigenValues[j]) {
                std::swap(eigenValues[i], eigenValues[j]);
                for (int k = 0; k < 3; ++k) {
                    std::swap(eigenVectors[k][i], eigenVectors[k][j]);
                }
            }
        }
    }
}

// ==================== 高级PCA法线估计 ====================
static Normal3D pcaNormal(const std::vector<Point3D>& points, int idx, int k, const SpatialHash* spatialHash = nullptr) {
    const Point3D& center = points[idx];
    std::vector<int> neighbors;
    
    if (spatialHash) {
        // 使用空间哈希快速搜索
        float radius = PoissonConfig::instance().sdfSearchRadius;
        auto candidates = spatialHash->queryRadius(center.x, center.y, center.z, radius);
        
        std::vector<std::pair<float, int>> dists;
        for (int i : candidates) {
            if (i == idx) continue;
            float dx = points[i].x - center.x;
            float dy = points[i].y - center.y;
            float dz = points[i].z - center.z;
            float dist2 = dx*dx + dy*dy + dz*dz;
            dists.emplace_back(dist2, i);
        }
        
        std::sort(dists.begin(), dists.end());
        int useCount = std::min(k, static_cast<int>(dists.size()));
        for (int i = 0; i < useCount; ++i) {
            neighbors.push_back(dists[i].second);
        }
    } else {
        // 原始的暴力搜索（带优化）
        std::vector<std::pair<float, int>> dists;
        dists.reserve(points.size());
        
        for (int i = 0; i < points.size(); ++i) {
            if (i == idx) continue;
            float dx = points[i].x - center.x;
            float dy = points[i].y - center.y;
            float dz = points[i].z - center.z;
            float dist2 = dx*dx + dy*dy + dz*dz;
            dists.emplace_back(dist2, i);
        }
        
        std::nth_element(dists.begin(), dists.begin() + k, dists.end());
        for (int i = 0; i < k && i < dists.size(); ++i) {
            neighbors.push_back(dists[i].second);
        }
    }
    
    if (neighbors.size() < 3) {
        return Normal3D(0, 0, -1);
    }
    
    // 计算质心
    float cx = 0, cy = 0, cz = 0;
    for (int i : neighbors) {
        cx += points[i].x;
        cy += points[i].y;
        cz += points[i].z;
    }
    cx /= neighbors.size();
    cy /= neighbors.size();
    cz /= neighbors.size();
    
    // 构建协方差矩阵
    Matrix3x3 cov;
    cov.setZero();
    
    for (int i : neighbors) {
        float dx = points[i].x - cx;
        float dy = points[i].y - cy;
        float dz = points[i].z - cz;
        
        cov.data[0][0] += dx * dx;
        cov.data[0][1] += dx * dy;
        cov.data[0][2] += dx * dz;
        cov.data[1][0] += dy * dx;
        cov.data[1][1] += dy * dy;
        cov.data[1][2] += dy * dz;
        cov.data[2][0] += dz * dx;
        cov.data[2][1] += dz * dy;
        cov.data[2][2] += dz * dz;
    }
    cov /= neighbors.size();
    
    // Jacobi特征值分解
    float eigenValues[3];
    float eigenVectors[3][3];
    jacobiEigenDecomposition(cov, eigenValues, eigenVectors);
    
    // 最小特征值对应的特征向量（第0列）
    Normal3D n(eigenVectors[0][0], eigenVectors[1][0], eigenVectors[2][0]);
    n.normalize();
    
    // 法向量朝向相机
    float vx = center.x, vy = center.y, vz = center.z;
    if (vx*n.nx + vy*n.ny + vz*n.nz > 0) {
        n.nx = -n.nx;
        n.ny = -n.ny;
        n.nz = -n.nz;
    }
    
    return n;
}

// ==================== 高级深度图转点云+法线 ====================
static std::vector<PointWithNormal> depthToPointsPCA(const DepthImage& depth, const CameraIntrinsics& K, 
                                                     float depthThreshold, int step, int normalNeighbors) {
    std::cout << "[INFO] 高级深度图转换 - 开始处理..." << std::endl;
    
    // 第一遍：生成所有有效点
    std::vector<Point3D> pts;
    std::vector<int> validIndices;
    
    int validDepthCount = 0;
    int totalPixels = 0;
    
    for (int y = step; y < depth.rows - step; y += step) {
        for (int x = step; x < depth.cols - step; x += step) {
            totalPixels++;
            float d = depth.at(y, x);
            
            if (d <= 0 || d > depthThreshold || std::isnan(d) || std::isinf(d)) continue;
            
            // 检查邻域深度连续性
            float neighbors[4] = {
                depth.at(y, x - step),
                depth.at(y, x + step),
                depth.at(y - step, x),
                depth.at(y + step, x)
            };
            
            bool validNeighbors = true;
            float depthDiffThreshold = d * PoissonConfig::instance().depthDiffThreshold;
            
            for (int i = 0; i < 4; ++i) {
                if (neighbors[i] <= 0 || std::isnan(neighbors[i]) || 
                    std::abs(neighbors[i] - d) > depthDiffThreshold) {
                    validNeighbors = false;
                    break;
                }
            }
            
            if (!validNeighbors) continue;
            
            float X = (x - K.cx) * d / K.fx;
            float Y = (y - K.cy) * d / K.fy;
            float Z = d;
            
            pts.emplace_back(X, Y, Z);
            validIndices.push_back(y * depth.cols + x);
            validDepthCount++;
        }
    }
    
    std::cout << "[DEBUG] 深度图统计: 总像素=" << totalPixels 
              << ", 有效深度=" << validDepthCount 
              << ", 最终点数=" << pts.size() << std::endl;
    
    if (pts.size() < 100) {
        std::cerr << "[ERROR] 有效点数太少，无法进行重建" << std::endl;
        return {};
    }
    
    // 构建空间哈希表以加速邻域搜索
    std::cout << "[INFO] 构建空间索引..." << std::endl;
    SpatialHash spatialHash(PoissonConfig::instance().voxelSize * 2.0f);
    for (size_t i = 0; i < pts.size(); ++i) {
        spatialHash.insert(i, pts[i].x, pts[i].y, pts[i].z);
    }
    
    // 第二遍：并行计算法线
    std::cout << "[INFO] 计算法线（并行）..." << std::endl;
    std::vector<PointWithNormal> result;
    result.reserve(pts.size());

#ifdef _OPENMP
    if (PoissonConfig::instance().useParallel) {
        // 创建临时存储用于并行处理
        std::vector<std::vector<PointWithNormal>> threadResults(omp_get_max_threads());
        
        #pragma omp parallel
        {
            int threadId = omp_get_thread_num();
            int numThreads = omp_get_num_threads();
            int chunkSize = (pts.size() + numThreads - 1) / numThreads;
            int start = threadId * chunkSize;
            int end = std::min(start + chunkSize, static_cast<int>(pts.size()));
            
            threadResults[threadId].reserve(end - start);
            
            for (int i = start; i < end; ++i) {
                Normal3D n = pcaNormal(pts, i, normalNeighbors, &spatialHash);
                threadResults[threadId].emplace_back(pts[i], n);
            }
        }
        
        // 合并结果
        for (const auto& threadResult : threadResults) {
            result.insert(result.end(), threadResult.begin(), threadResult.end());
        }
    } else
#endif
    {
        for (size_t i = 0; i < pts.size(); ++i) {
            Normal3D n = pcaNormal(pts, i, normalNeighbors, &spatialHash);
            result.emplace_back(pts[i], n);
            
            if (PoissonConfig::instance().enableDebug && i % 1000 == 0) {
                std::cout << "[DEBUG] 处理进度: " << i << "/" << pts.size() << std::endl;
            }
        }
    }
    
    // 第二遍：并行计算法线
    std::cout << "[INFO] 计算法线（并行）..." << std::endl;
    // std::vector<PointWithNormal> result;
    result.reserve(pts.size());

#ifdef _OPENMP
    if (PoissonConfig::instance().useParallel) {
        // 创建临时存储用于并行处理
        std::vector<std::vector<PointWithNormal>> threadResults(omp_get_max_threads());
        
        #pragma omp parallel
        {
            int threadId = omp_get_thread_num();
            int numThreads = omp_get_num_threads();
            int chunkSize = (pts.size() + numThreads - 1) / numThreads;
            int start = threadId * chunkSize;
            int end = std::min(start + chunkSize, static_cast<int>(pts.size()));
            
            threadResults[threadId].reserve(end - start);
            
            for (int i = start; i < end; ++i) {
                Normal3D n = pcaNormal(pts, i, normalNeighbors, &spatialHash);
                threadResults[threadId].emplace_back(pts[i], n);
            }
        }
        
        // 合并结果
        for (const auto& threadResult : threadResults) {
            result.insert(result.end(), threadResult.begin(), threadResult.end());
        }
    } else
#endif
    {
        for (size_t i = 0; i < pts.size(); ++i) {
            Normal3D n = pcaNormal(pts, i, normalNeighbors, &spatialHash);
            result.emplace_back(pts[i], n);
            
            if (PoissonConfig::instance().enableDebug && i % 1000 == 0) {
                std::cout << "[DEBUG] 处理进度: " << i << "/" << pts.size() << std::endl;
            }
        }
    }
    
    std::cout << "[INFO] 法线计算完成，生成 " << result.size() << " 个带法线的点" << std::endl;
    return result;
}

// ==================== SDF计算 ====================
static float calculateSDF(float worldX, float worldY, float worldZ, 
                         const std::vector<PointWithNormal>& pointsWithNormals,
                         const SpatialHash& spatialHash) {
    
    auto candidates = spatialHash.queryRadius(worldX, worldY, worldZ, 
                                            PoissonConfig::instance().sdfSearchRadius);
    
    if (candidates.empty()) return 0.0f;
    
    float weightedSDF = 0.0f;
    float totalWeight = 0.0f;
    float sigma = PoissonConfig::instance().sdfSigma;
    int maxNeighbors = PoissonConfig::instance().sdfNeighbors;
    
    // 计算距离并排序
    std::vector<std::pair<float, int>> dists;
    for (int i : candidates) {
        if (i < 0 || i >= pointsWithNormals.size()) continue;
        
        const auto& point = pointsWithNormals[i];
        float dx = worldX - point.point.x;
        float dy = worldY - point.point.y;
        float dz = worldZ - point.point.z;
        float distance = std::sqrt(dx*dx + dy*dy + dz*dz);
        dists.emplace_back(distance, i);
    }
    
    std::sort(dists.begin(), dists.end());
    int useCount = std::min(maxNeighbors, static_cast<int>(dists.size()));
    
    for (int i = 0; i < useCount; ++i) {
        const auto& point = pointsWithNormals[dists[i].second];
        
        float dx = worldX - point.point.x;
        float dy = worldY - point.point.y;
        float dz = worldZ - point.point.z;
        float distance = dists[i].first;
        
        // 有向距离
        float signedDist = dx * point.normal.nx + dy * point.normal.ny + dz * point.normal.nz;
        
        // 高斯权重
        float weight = std::exp(-distance * distance / (2.0f * sigma * sigma));
        
        weightedSDF += weight * signedDist;
        totalWeight += weight;
    }
    
    return totalWeight > 1e-6f ? weightedSDF / totalWeight : 0.0f;
}

// ==================== Marching Cubes风格的网格生成 ====================
static void advancedTriangulation(const std::vector<PointWithNormal>& pointsWithNormals,
                                 const DepthImage& depth, const ColorImage& color, 
                                 const CameraIntrinsics& K, Mesh& mesh) {
    
    std::cout << "[INFO] 高级三角剖分开始..." << std::endl;
    
    // 计算包围盒
    float minX = std::numeric_limits<float>::max();
    float maxX = std::numeric_limits<float>::lowest();
    float minY = std::numeric_limits<float>::max();
    float maxY = std::numeric_limits<float>::lowest();
    float minZ = std::numeric_limits<float>::max();
    float maxZ = std::numeric_limits<float>::lowest();
    
    for (const auto& pwn : pointsWithNormals) {
        minX = std::min(minX, pwn.point.x);
        maxX = std::max(maxX, pwn.point.x);
        minY = std::min(minY, pwn.point.y);
        maxY = std::max(maxY, pwn.point.y);
        minZ = std::min(minZ, pwn.point.z);
        maxZ = std::max(maxZ, pwn.point.z);
    }
    
    // 构建SDF空间哈希
    SpatialHash sdfHash(PoissonConfig::instance().voxelSize);
    for (size_t i = 0; i < pointsWithNormals.size(); ++i) {
        const auto& p = pointsWithNormals[i].point;
        sdfHash.insert(i, p.x, p.y, p.z);
    }
    
    float voxelSize = PoissonConfig::instance().voxelSize;
    int gridSizeX = std::min(PoissonConfig::instance().maxGridSize, 
                            static_cast<int>((maxX - minX) / voxelSize) + 1);
    int gridSizeY = std::min(PoissonConfig::instance().maxGridSize,
                            static_cast<int>((maxY - minY) / voxelSize) + 1);
    int gridSizeZ = std::min(PoissonConfig::instance().maxGridSize,
                            static_cast<int>((maxZ - minZ) / voxelSize) + 1);
    
    std::cout << "[DEBUG] 网格尺寸: " << gridSizeX << "x" << gridSizeY << "x" << gridSizeZ << std::endl;
    
    // 简化的等值面提取（基于原始点的重新三角化）
    std::map<std::pair<int,int>, int> vertexMap;
    
    // 基于深度图的改进栅格三角剖分
    int step = PoissonConfig::instance().triangulationStep;
    for (int y = 0; y < depth.rows - step; y += step) {
        for (int x = 0; x < depth.cols - step; x += step) {
            // 获取四个角点的深度
            float d00 = depth.at(y, x);
            float d01 = depth.at(y, x + step);
            float d10 = depth.at(y + step, x);
            float d11 = depth.at(y + step, x + step);
            
            // 检查所有深度值有效
            if (d00 <= 0 || d01 <= 0 || d10 <= 0 || d11 <= 0 ||
                d00 > PoissonConfig::instance().depthThreshold ||
                d01 > PoissonConfig::instance().depthThreshold ||
                d10 > PoissonConfig::instance().depthThreshold ||
                d11 > PoissonConfig::instance().depthThreshold) {
                continue;
            }
            
            // 计算四个3D顶点
            auto addVertex = [&](int px, int py, float depth) -> int {
                auto key = std::make_pair(px, py);
                auto it = vertexMap.find(key);
                if (it != vertexMap.end()) {
                    return it->second;
                }
                
                float X = (px - K.cx) * depth / K.fx;
                float Y = (py - K.cy) * depth / K.fy;
                float Z = depth;
                
                uint8_t r = 128, g = 128, b = 128;  // 默认灰色
                
                // 根据配置决定是否使用颜色
                if (PoissonConfig::instance().useColor && 
                    py < color.rows && px < color.cols) {
                    const uint8_t* pix = color.pixel(py, px);
                    r = pix[0]; g = pix[1]; b = pix[2];
                }
                
                int vertexIndex = mesh.vertices.size();
                mesh.vertices.emplace_back(X, Y, Z, r, g, b);
                vertexMap[key] = vertexIndex;
                return vertexIndex;
            };
            
            int v00 = addVertex(x, y, d00);
            int v01 = addVertex(x + step, y, d01);
            int v10 = addVertex(x, y + step, d10);
            int v11 = addVertex(x + step, y + step, d11);
            
            // 检查深度连续性，避免跨越深度断层的三角形
            float maxDepthDiff = std::max({
                std::abs(d00 - d01), std::abs(d00 - d10), std::abs(d01 - d11),
                std::abs(d10 - d11), std::abs(d00 - d11), std::abs(d01 - d10)
            });
            
            float avgDepth = (d00 + d01 + d10 + d11) / 4.0f;
            
            // 使用配置的深度差异阈值和最大深度差
            bool depthContinuous = (maxDepthDiff < avgDepth * PoissonConfig::instance().depthDiffThreshold) &&
                                 (maxDepthDiff < PoissonConfig::instance().maxDepthDifference);
            
            if (depthContinuous) {
                // 生成两个三角形
                mesh.faces.emplace_back(v00, v01, v10);
                mesh.faces.emplace_back(v01, v11, v10);
            }
        }
    }
    
    std::cout << "[INFO] 高级三角剖分完成: " << mesh.vertices.size() 
              << " 顶点, " << mesh.faces.size() << " 面" << std::endl;
}

// ==================== 主Poisson重建函数 ====================
Mesh poissonReconstruction(const DepthImage& depth, const ColorImage& color, const CameraIntrinsics& K, 
                          float depthThreshold, int step, int normalNeighbors) {
    Mesh mesh;
    using namespace std::chrono;
    
    // 更新全局配置
    PoissonConfig::instance().depthThreshold = depthThreshold;
    PoissonConfig::instance().triangulationStep = step;
    PoissonConfig::instance().normalNeighbors = normalNeighbors;
    
    // 显示完整参数配置
    PoissonConfig::instance().printConfig();
    
    std::cout << "==================== 高级Poisson重建开始 ====================" << std::endl;
    std::cout << "[INFO] 运行时参数:" << std::endl;
    std::cout << "  深度图尺寸: " << depth.rows << "x" << depth.cols << std::endl;
    std::cout << "  颜色图尺寸: " << color.rows << "x" << color.cols << std::endl;
    std::cout << "  相机内参: fx=" << K.fx << ", fy=" << K.fy << ", cx=" << K.cx << ", cy=" << K.cy << std::endl;
#ifdef _OPENMP
    if (PoissonConfig::instance().useParallel) {
        std::cout << "  OpenMP线程数: " << omp_get_max_threads() << std::endl;
    }
#endif
    
    auto t0 = high_resolution_clock::now();
    
    // 1. 高级点云+法线生成
    std::cout << "\n[步骤1] 点云+法线生成..." << std::endl;
    auto t1 = high_resolution_clock::now();
    auto points = depthToPointsPCA(depth, K, depthThreshold, step, normalNeighbors);
    auto t2 = high_resolution_clock::now();
    std::cout << "[完成] 点云+法线生成, 用时: " << duration_cast<milliseconds>(t2-t1).count() 
              << " ms, 点数: " << points.size() << std::endl;
    
    if (points.size() < 100) {
        std::cerr << "[ERROR] 点数不足，重建失败" << std::endl;
        return mesh;
    }
    
    // 2. 高级三角剖分
    std::cout << "\n[步骤2] 高级三角剖分..." << std::endl;
    auto t3 = high_resolution_clock::now();
    advancedTriangulation(points, depth, color, K, mesh);
    auto t4 = high_resolution_clock::now();
    std::cout << "[完成] 三角剖分, 用时: " << duration_cast<milliseconds>(t4-t3).count() 
              << " ms, 顶点: " << mesh.vertices.size() << " 面: " << mesh.faces.size() << std::endl;
    
    auto t5 = high_resolution_clock::now();
    std::cout << "\n==================== 重建完成 ====================" << std::endl;
    std::cout << "[总计] 用时: " << duration_cast<milliseconds>(t5-t0).count() << " ms" << std::endl;
    std::cout << "[结果] 顶点数: " << mesh.vertices.size() << ", 面数: " << mesh.faces.size() << std::endl;
    std::cout << "======================================================\n" << std::endl;
    
    return mesh;
}
