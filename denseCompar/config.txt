# DenseCompar 配置文件
# 支持 key=value 格式，注释以#开头

# ========== 视差计算参数 ==========
# numDisparities: 最大视差范围（必须为16的倍数），影响深度精度和计算速度。推荐128~256。
numDisparities=256
# blockSize: 匹配窗口大小（奇数），越大越平滑但细节损失，越小细节多但噪声大。推荐5~15。
blockSize=11

# ========== 路径参数 ==========
# leftDir/rightDir: 左右图像文件夹路径（末尾加/）
# cameraDir: 相机内参文件夹路径
# outputDir: 输出结果文件夹路径
leftDir=data/left/
rightDir=data/right/
cameraDir=data/camera/
outputDir=output/

# ========== 相机畸变参数 ==========
# distCoeffs: 相机畸变系数（k1,k2,p1,p2,k3），通常为5个浮点数。无畸变可全0。
distCoeffs=0,0,0,0,0

# ========== denseMatching/SGBM参数 ==========
# sgbmP1: SGBM平滑项P1，较小值更敏感，推荐1-10。P1变大，视差图更平滑但细节丢失，变小则噪声增多但细节更丰富。
sgbmP1=1
# sgbmP2: SGBM平滑项P2，较大值更平滑，推荐4-40。P2变大，能抑制错误匹配但可能丢失边缘，变小则边缘更锐利但易出错。
sgbmP2=4
# preFilterCap: 预滤波器上限，推荐15-63。变大可增强对比度但易引入伪影，变小则对比度低但更平滑。
preFilterCap=31
# uniquenessRatio: 唯一性比率，越大越严格，推荐5-50。变大可减少误匹配但可能导致更多无效区域，变小则视差更密集但误差增多。
uniquenessRatio=50
# speckleWindowSize: 孤立斑点窗口大小，推荐0-200。变大可去除更多小噪点但可能误删细节，变小则噪点增多。
speckleWindowSize=100
# speckleRange: 孤立斑点视差范围，推荐0-32。变大可去除更多异常视差但可能误删，变小则保留更多细节但噪点增多。
speckleRange=8
# disp12MaxDiff: 左右一致性最大容差，推荐1-10。变大可保留更多视差但误差增多，变小则更严格但无效区域增多。
disp12MaxDiff=5

# ========== 特征匹配和位姿估计参数 ==========
# use8pointAlgorithm: 是否使用8point算法估计R和t参数，true=使用，false=使用硬编码默认值
use8pointAlgorithm=true
# featureType: 特征检测器类型，可选ORB/SIFT/SURF
featureType=SIFT

# ========== ELAS算法参数（已注释，使用SGBM参数值） ==========
# 注意：以下ELAS参数已被注释，ELAS算法将使用对应的SGBM参数值
# elasPreset: ELAS预设配置，可选ROBOTICS/MIDDLEBURY/CUSTOM
elasPreset=CUSTOM

# ELAS视差范围参数（对应SGBM的numDisparities=256）
# elasDispMin=0
# elasDispMax=255
elasDispMin=0
elasDispMax=256

# ELAS支持点检测参数
# elasSupportThreshold=0.85  # 对应SGBM的uniquenessRatio=50
# elasSupportTexture=10      # 对应SGBM的preFilterCap=31
# elasCandidateStepsize=5
# elasAddCorners=true
elasSupportThreshold=0.50  # 对应SGBM的uniquenessRatio=50 (50/100)
elasSupportTexture=31      # 对应SGBM的preFilterCap=31
elasCandidateStepsize=5
elasAddCorners=false

# ELAS一致性检查参数

elasInconWindowSize=5
elasInconThreshold=5
elasInconMinSupport=5

# ELAS网格和插值参数
elasGridSize=20
elasIpolGapWidth=3

# ELAS概率模型参数
elasBeta=0.02
elasGamma=3.0
elasSigma=1.0
elasSradius=2.0

# ELAS匹配参数
# elasMatchTexture=1
# elasLrThreshold=2          # 对应SGBM的disp12MaxDiff=5
elasMatchTexture=31        # 对应SGBM的preFilterCap=31
elasLrThreshold=5          # 对应SGBM的disp12MaxDiff=5

# ELAS后处理参数
# elasSpeckleSimThreshold=1.0
# elasSpeckleSize=200        # 对应SGBM的speckleWindowSize=100
# elasFilterMedian=false
# elasFilterAdaptiveMean=true
# elasPostprocessOnlyLeft=true

elasSpeckleSimThreshold=1.0
elasSpeckleSize=100        # 对应SGBM的speckleWindowSize=100
elasFilterMedian=false
elasFilterAdaptiveMean=true
elasPostprocessOnlyLeft=true

# ELAS性能参数
elasSubsampling=false
